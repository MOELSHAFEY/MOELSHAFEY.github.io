// Clean Architecture: Pure Domain Contract
abstract class UseCase<Type, Params> {
  Future<Either<Failure, Type>> call(Params params);
}

// Production BLoC State Management
class PortalBloc extends Bloc<PortalEvent, PortalState> {
  final GetAcademicNews getNews;

  PortalBloc({required this.getNews}) 
      : super(PortalInitial()) {
    on<FetchNews>((event, emit) async {
      emit(PortalLoading());
      final result = await getNews(NoParams());
      result.fold(
        (failure) => emit(PortalError(failure.message)),
        (news) => emit(PortalLoaded(news)),
      );
    });
  }
}
